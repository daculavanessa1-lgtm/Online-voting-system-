const banner = document.getElementById('banner'); const userBar = document.getElementById('userBar'); const authSection = document.getElementById('authSection'); const voteStatusText = document.getElementById('voteStatusText'); const candidatesList = document.getElementById('candidatesList'); const resultsList = document.getElementById('resultsList'); const totalVotesLabel = document.getElementById('totalVotesLabel'); let currentState = { user: null, hasVoted: false, votedFor: null }; let currentCandidates = []; function showBanner(message, type = 'success') { banner.textContent = message; banner.className = `alert ${type === 'error' ? 'alert-error' : 'alert-success'}`; } function clearBanner() { banner.className = 'alert hidden'; banner.textContent = ''; } async function api(url, options = {}) { const res = await fetch(url, { headers: { 'Content-Type': 'application/json' }, ...options }); const data = await res.json(); if (!res.ok) { throw new Error(data.error || 'Request failed'); } return data; } function renderAuth() { if (currentState.user) { authSection.classList.add('hidden'); userBar.innerHTML = ` Signed in as ${escapeHtml(currentState.user.name)} `; document.getElementById('logoutBtn').addEventListener('click', async () => { await api('/api/logout', { method: 'POST' }); clearBanner(); await refreshAll(); }); } else { authSection.classList.remove('hidden'); userBar.innerHTML = `Not signed in`; } } function renderCandidates() { if (!currentState.user) { voteStatusText.textContent = 'Log in above to cast your ballot. Each account gets one vote.'; } else if (currentState.hasVoted) { voteStatusText.textContent = `You have already voted for ${currentState.votedFor}.`; } else { voteStatusText.textContent = 'Select a candidate below to submit your single vote.'; } const canVote = Boolean(currentState.user) && !currentState.hasVoted; candidatesList.innerHTML = currentCandidates .map( (c) => `
${escapeHtml(c.party)}

${escapeHtml(c.name)}

${escapeHtml(c.bio || '')}

` ) .join(''); candidatesList.querySelectorAll('button[data-vote-id]').forEach((btn) => { btn.addEventListener('click', async () => { const candidateId = Number(btn.dataset.voteId); try { const res = await api('/api/vote', { method: 'POST', body: JSON.stringify({ candidateId }) }); showBanner(res.message, 'success'); await refreshAll(); } catch (err) { showBanner(err.message, 'error'); } }); }); } function renderResults(data) { totalVotesLabel.textContent = `${data.totalVotes} total ${data.totalVotes === 1 ? 'vote' : 'votes'}`; resultsList.innerHTML = data.candidates .map((c) => { const pct = data.totalVotes > 0 ? Math.round((c.vote_count / data.totalVotes) * 100) : 0; return `
${escapeHtml(c.name)} (${escapeHtml(c.party)}) ${c.vote_count} (${pct}%)
`; }) .join(''); } function escapeHtml(str) { return String(str) .replaceAll('&', '&') .replaceAll('<', '<') .replaceAll('>', '>') .replaceAll('"', '"'); } async function refreshAll() { const [me, results] = await Promise.all([api('/api/me'), api('/api/results')]); currentState = me; currentCandidates = results.candidates; renderAuth(); renderCandidates(); renderResults(results); } document.getElementById('loginForm').addEventListener('submit', async (e) => { e.preventDefault(); clearBanner(); try { await api('/api/login', { method: 'POST', body: JSON.stringify({ email: document.getElementById('loginEmail').value, password: document.getElementById('loginPassword').value }) }); showBanner('Logged in successfully.', 'success'); await refreshAll(); } catch (err) { showBanner(err.message, 'error'); } }); document.getElementById('registerForm').addEventListener('submit', async (e) => { e.preventDefault(); clearBanner(); try { await api('/api/register', { method: 'POST', body: JSON.stringify({ name: document.getElementById('regName').value, email: document.getElementById('regEmail').value, password: document.getElementById('regPassword').value }) }); showBanner('Account created and logged in.', 'success'); await refreshAll(); } catch (err) { showBanner(err.message, 'error'); } }); refreshAll().catch((err) => showBanner(err.message, 'error'));
